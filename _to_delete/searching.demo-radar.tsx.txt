/** Tactical Circular Radar Dish Grid with Pitch Rings and Degree Marks */
function TacticalRadarGrid({ size }: { size: number }): React.ReactNode {
  const r = size / 2;
  return (
    <Svg width={size} height={size} viewBox={`0 0 ${size} ${size}`} style={StyleSheet.absoluteFill}>
      <Defs>
        <RadialGradient id="radarDishBg" cx="50%" cy="50%" r="50%">
          <Stop offset="0%" stopColor="#082014" stopOpacity="1" />
          <Stop offset="65%" stopColor="#04120B" stopOpacity="1" />
          <Stop offset="100%" stopColor="#020805" stopOpacity="1" />
        </RadialGradient>
        <RadialGradient id="dishCenterAura" cx="50%" cy="50%" r="50%">
          <Stop offset="0%" stopColor="#10B981" stopOpacity="0.3" />
          <Stop offset="50%" stopColor="#10B981" stopOpacity="0.08" />
          <Stop offset="100%" stopColor="#10B981" stopOpacity="0" />
        </RadialGradient>
      </Defs>

      {/* Deep Turf Background */}
      <Circle
        cx={r}
        cy={r}
        r={r - 2}
        fill="url(#radarDishBg)"
        stroke="rgba(16, 185, 129, 0.4)"
        strokeWidth="1.5"
      />

      {/* Center Aura */}
      <Circle cx={r} cy={r} r={r * 0.8} fill="url(#dishCenterAura)" />

      {/* Crosshair Axes */}
      <Line
        x1={10}
        y1={r}
        x2={size - 10}
        y2={r}
        stroke="rgba(16, 185, 129, 0.25)"
        strokeWidth="1"
        strokeDasharray="4,4"
      />
      <Line
        x1={r}
        y1={10}
        x2={r}
        y2={size - 10}
        stroke="rgba(16, 185, 129, 0.25)"
        strokeWidth="1"
        strokeDasharray="4,4"
      />

      {/* Diagonal Grid Lines */}
      <Line
        x1={r * 0.3}
        y1={r * 0.3}
        x2={size - r * 0.3}
        y2={size - r * 0.3}
        stroke="rgba(16, 185, 129, 0.12)"
        strokeWidth="0.8"
        strokeDasharray="2,4"
      />
      <Line
        x1={size - r * 0.3}
        y1={r * 0.3}

        y2={size - r * 0.3}
        stroke="rgba(16, 185, 129, 0.12)"

      <Circle cx={r} cy={r} r={r * 0.24} stroke="rgba(16, 185, 129, 0.6)" strokeWidth="1.2" fill="none" />

    </Svg>
  );
}

/** Sweeping Radar Cone Beam */
function TacticalRadarBeam({ size }: { size: number }): React.ReactNode {
  const r = size / 2;
  const rotation = useSharedValue(0);

  useEffect(() => {
    rotation.value = withRepeat(
      withTiming(360, {
        duration: SWEEP_MS,
        easing: Easing.linear,
      }),
      -1,
      false,
    );
  }, [rotation]);

  const animatedStyle = useAnimatedStyle(() => ({
    transform: [{ rotate: `${rotation.value}deg` }],
  }));

  return (
    <View style={StyleSheet.absoluteFill} pointerEvents="none">
      <Animated.View style={[{ width: size, height: size }, animatedStyle]}>
        <Svg width={size} height={size} viewBox={`0 0 ${size} ${size}`}>
          <Defs>
            <SvgLinearGradient id="tacticalSweep" x1="0%" y1="0%" x2="100%" y2="0%">
              <Stop offset="0%" stopColor="#10B981" stopOpacity="0" />
              <Stop offset="60%" stopColor="#10B981" stopOpacity="0.18" />
              <Stop offset="100%" stopColor="#34D399" stopOpacity="0.45" />
            </SvgLinearGradient>
          </Defs>

          {/* Sweeping Cone */}
          <Path
            d={`M ${r} ${r} L ${r + r * 0.94 * Math.sin(-0.55)} ${r - r * 0.94 * Math.cos(-0.55)} A ${r * 0.94} ${r * 0.94} 0 0 1 ${r} ${r * 0.06} Z`}
            fill="url(#tacticalSweep)"
          />

          {/* Laser Line */}
          <Line
            x1={r}
            y1={r}
            x2={r}
            y2={r * 0.06}
            stroke="#34D399"
            strokeWidth="2.2"
            strokeLinecap="round"
          />

          {/* Laser Tip Beacon */}
          <Circle cx={r} cy={r * 0.06} r="4.5" fill="#FFFFFF" stroke="#10B981" strokeWidth="2" />
        </Svg>
      </Animated.View>
    </View>
  );
}

/** Expanding Pulse Waves */
function TacticalPulseRing({ size, delay = 0 }: { size: number; delay?: number }): React.ReactNode {
  const scale = useSharedValue(0.2);
  const opacity = useSharedValue(0.8);

  useEffect(() => {
    const timer = setTimeout(() => {
      scale.value = withRepeat(
        withTiming(1.02, { duration: SWEEP_MS, easing: Easing.out(Easing.ease) }),
        -1,
        false,
      );
      opacity.value = withRepeat(
        withTiming(0, { duration: SWEEP_MS, easing: Easing.out(Easing.ease) }),
        -1,
        false,
      );
    }, delay);
    return () => clearTimeout(timer);
  }, [scale, opacity, delay]);

  const animatedStyle = useAnimatedStyle(() => ({
    transform: [{ scale: scale.value }],
    opacity: opacity.value,
  }));

  return (
    <Animated.View
      style={[
        {
          position: 'absolute',
          width: size,
          height: size,
          borderRadius: 999,
          borderWidth: 1.5,
          borderColor: '#10B981',
          shadowColor: '#10B981',
          shadowOffset: { width: 0, height: 0 },
          shadowOpacity: 0.6,
          shadowRadius: 8,
        },
        animatedStyle,
      ]}
      pointerEvents="none"
    />
  );
}
